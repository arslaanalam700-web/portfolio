# Production Data Pipeline & Dynamic Metric Aggregation
import pandas as pd
import numpy as np

def compute_terminal_kpis(df: pd.DataFrame) -> pd.DataFrame:
    df['congestion_idx'] = (df['pax_flow'] / df['capacity']) * 100
    df['peak_oli'] = np.where(df['congestion_idx'] > 85, 1.25, 0.85)
    return df.groupby('terminal').agg({'congestion_idx': 'mean', 'peak_oli': 'max'})
